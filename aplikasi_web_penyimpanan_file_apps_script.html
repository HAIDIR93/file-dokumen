// BACKEND GOOGLE APPS SCRIPT (Code.gs)
// Pasang kode ini pada Apps Script Editor (script.google.com)
// untuk menghubungkan UI ini ke Google Drive asli milik Anda.
// ============================================================

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('GAS DocVault - Drive Storage Web App')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// Memuat daftar seluruh berkas dari Google Drive
function getDriveFiles(folderId) {
  try {
    var folder = folderId ? DriveApp.getFolderById(folderId) : DriveApp.getRootFolder();
    var files = folder.getFiles();
    var list = [];

    while (files.hasNext()) {
      var f = files.next();
      list.push({
        id: f.getId(),
        name: f.getName(),
        size: formatBytes(f.getSize()),
        mimeType: f.getMimeType(),
        url: f.getUrl(),
        dateUpdated: Utilities.formatDate(f.getLastUpdated(), Session.getScriptTimeZone(), "yyyy-MM-dd")
      });
    }
    return JSON.stringify({ success: true, data: list });
  } catch (err) {
    return JSON.stringify({ success: false, error: err.toString() });
  }
}

// Mengunggah file baru berformat Base64 ke Google Drive
function uploadFileToDrive(base64Data, fileName, mimeType, targetFolderId) {
  try {
    var folder = targetFolderId ? DriveApp.getFolderById(targetFolderId) : DriveApp.getRootFolder();
    var bytes = Utilities.base64Decode(base64Data.split(',')[1]);
    var blob = Utilities.newBlob(bytes, mimeType, fileName);
    var newFile = folder.createFile(blob);
    
    return {
      success: true,
      fileId: newFile.getId(),
      fileUrl: newFile.getUrl()
    };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function formatBytes(bytes) {
  if (bytes === 0) return '0 Bytes';
  var k = 1024, sizes = ['Bytes', 'KB', 'MB', 'GB'];
  var i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
}
